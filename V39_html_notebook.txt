  <!-- 🟢 CAPA SCRIPT (LOGICA DE INTEGRACIÓN CON GOOGLE APPS SCRIPT) -->
  <script>
    // Variables de Estado Local
    let pochtecas = [];
    let processingImage = false;

    // Inicializar operaciones al cargar la página
    window.addEventListener('load', () => {
      actualizarTodo();
    });

    function addLog(msg, error = false) {
      const logs = document.getElementById('live-logs');
      const time = new Date().toLocaleTimeString("es-MX");
      const span = document.createElement('div');
      span.className = error ? "text-rose-400 font-semibold" : "text-slate-300";
      span.innerHTML = `[${time}] ${msg}`;
      logs.appendChild(span);
      logs.scrollTop = logs.scrollHeight;
    }

    function actualizarTodo() {
      addLog("Sincronizando consola con el Amoxcalli...");
      cargarCatalogos();
      cargarMonitor();
      cargarCorreos();
    }

    function cargarCatalogos() {
      google.script.run
        .withSuccessHandler((data) => {
          pochtecas = data;
          addLog(`Catálogo de Pochtecas cargado con éxito (${data.length} usuarios).`);
          poblarSelectChoferes();
        })
        .withFailureHandler((err) => {
          addLog("Error al recuperar catálogo de usuarios: " + err.message, true);
        })
        .obtenerChoferesQRO();
    }

    function poblarSelectChoferes() {
      const select = document.getElementById('ia-chofer');
      select.innerHTML = '<option value="sin_asignar@arauto.express">-- Sin Asignar / Externo --</option>';
      pochtecas.forEach(p => {
        select.innerHTML += `<option value="${p.correo}">${p.nombre} (${p.correo})</option>`;
      });
    }

    function cargarMonitor() {
      google.script.run
        .withSuccessHandler((lista) => {
          const tbody = document.getElementById('tabla-monitor');
          if (lista.length === 0) {
            tbody.innerHTML = '<tr><td colspan="6" class="text-center py-8 text-slate-400">No hay recolecciones despachadas el día de hoy</td></tr>';
            return;
          }
          tbody.innerHTML = '';
          lista.forEach(item => {
            let badgeColor = "bg-amber-100 text-amber-800";
            if (item.estatus === "OK") badgeColor = "bg-emerald-100 text-emerald-800";
            if (item.estatus === "NH" || item.estatus === "BA") badgeColor = "bg-rose-100 text-rose-800";

            tbody.innerHTML += `
              <tr class="hover:bg-slate-50 transition">
                <td class="px-4 py-3 font-semibold text-slate-900">${item.id_pu}</td>
                <td class="px-4 py-3 text-slate-500">${item.id_booking}</td>
                <td class="px-4 py-3 font-medium text-slate-700">${item.remitente}</td>
                <td class="px-4 py-3 text-slate-500">${item.cp || "N/A"}</td>
                <td class="px-4 py-3 text-xs text-slate-600 font-semibold">${item.chofer.split('@')[0]}</td>
                <td class="px-4 py-3 text-xs"><span class="px-2 py-0.5 rounded-full font-bold ${badgeColor}">${item.estatus}</span></td>
              </tr>
            `;
          });
          addLog("Monitor de calle en vivo actualizado.");
        })
        .withFailureHandler((err) => {
          addLog("Error al refrescar monitor calle: " + err.message, true);
        })
        .obtenerPickupsActivosQRO();
    }

    function cargarCorreos() {
      google.script.run
        .withSuccessHandler((lista) => {
          const tbody = document.getElementById('tabla-correos');
          if (lista.length === 0) {
            tbody.innerHTML = '<tr><td colspan="5" class="text-center py-8 text-slate-400">No se encontraron correos pendientes en Gmail</td></tr>';
            return;
          }
          tbody.innerHTML = '';
          lista.forEach(mail => {
            const isQuery = mail.tipo === "QUERY";
            const typeBadge = isQuery 
              ? "bg-[#D4AF37] text-[#1e3f20] font-bold" 
              : "bg-indigo-100 text-indigo-800 font-bold";
            const unreadStyle = mail.isUnread ? "font-bold text-slate-900 bg-emerald-50/50" : "text-slate-600";
            
            tbody.innerHTML += `
              <tr class="hover:bg-slate-50 transition ${unreadStyle}">
                <td class="px-4 py-3 text-xs"><span class="px-2 py-0.5 rounded-full ${typeBadge}">${mail.tipo}</span></td>
                <td class="px-4 py-3 font-medium max-w-xs truncate">${mail.subject}</td>
                <td class="px-4 py-3 text-slate-500">${mail.sender}</td>
                <td class="px-4 py-3 text-slate-400 text-xs">${mail.date}</td>
                <td class="px-4 py-3 text-center">
                  <button onclick="forzarInyeccionMail('${mail.id}', '${mail.tipo}')" class="text-xs bg-[#1e3f20] hover:bg-[#142b15] text-white px-2.5 py-1 rounded font-semibold transition">
                    Inyectar
                  </button>
                </td>
              </tr>
            `;
          });
          addLog("Bandeja de correos de soporte de Gmail actualizada.");
        })
        .withFailureHandler((err) => {
          addLog("Error al recuperar correos de Gmail: " + err.message, true);
        })
        .buscarCorreosBatchQRO();
    }

    function forzarInyeccionMail(msgId, tipo) {
      addLog(`Detonando inyección asíncrona para ${tipo} (ID: ${msgId})...`);
      if (tipo === "QUERY") {
        google.script.run
          .withSuccessHandler((res) => {
            if (res.exito) {
              addLog(`ÉXITO: ${res.mensaje}`);
              actualizarTodo();
            } else {
              addLog(`FALLO: ${res.error || res.mensaje}`, true);
            }
          })
          .withFailureHandler((err) => {
            addLog("Error técnico en procesamiento asíncrono de Query: " + err.message, true);
          })
          .procesarQueriesQRO(msgId);
      } else {
        google.script.run
          .withSuccessHandler((res) => {
            if (res.exito) {
              addLog(`ÉXITO: ${res.mensaje}`);
              actualizarTodo();
            } else {
              addLog(`FALLO: ${res.error || res.mensaje}`, true);
            }
          })
          .withFailureHandler((err) => {
            addLog("Error técnico en procesamiento asíncrono de Batch: " + err.message, true);
          })
          .procesarBatchQRO(msgId);
      }
    }

    function recalcularChofer(cp) {
      if (!cp) return;
      google.script.run
        .withSuccessHandler((chofer) => {
          document.getElementById('ia-chofer').value = chofer;
          addLog(`Pochteca auto-asignado por C.P. ${cp}: ${chofer}`);
        })
        .autoAsignarPochtecaPorCP(cp);
    }

    // Convertidor de Imagen a Base64 e Integración con Gemini
    function procesarImagen(file) {
      if (!file) return;
      if (processingImage) return;

      processingImage = true;
      addLog(`Procesando imagen: ${file.name} (${Math.round(file.size/1024)} KB)...`);
      
      const reader = new FileReader();
      reader.readAsDataURL(file);
      reader.onload = function () {
        const base64Content = reader.result.split(',')[1];
        const mimeType = file.type;

        addLog("Enviando flujo visual a Gemini 3.6 Flash para interpretación...");
        
        google.script.run
          .withSuccessHandler((res) => {
            processingImage = false;
            if (res.exito) {
              addLog("Gemini interpretó con éxito la captura del Booking.");
              
              document.getElementById('ia-booking').value = res.datos.id_booking || "";
              document.getElementById('ia-remitente').value = res.datos.remitente || "";
              document.getElementById('ia-direccion').value = res.datos.direccion || "";
              document.getElementById('ia-cp').value = res.datos.cp || "";
              document.getElementById('ia-piezas').value = res.datos.piezas_estimadas || 1;
              document.getElementById('ia-tel').value = res.datos.telefono || "";
              document.getElementById('ia-chofer').value = res.datos.chofer_sugerido || "sin_asignar@arauto.express";

              document.getElementById('ia-preview').classList.remove('hidden');
              document.getElementById('btn-despacho').classList.remove('hidden');
            } else {
              addLog("Error en el lector de IA: " + res.error, true);
            }
          })
          .withFailureHandler((err) => {
            processingImage = false;
            addLog("Error de red con la IA de Gemini: " + err.message, true);
          })
          .analizarCapturaBookingConGemini(base64Content, mimeType);
      };
    }

    // Ejecuta el despacho final dual a rampa y calle
    function ejecutarDespacho() {
      addLog("Iniciando despacho y adquiriendo candado de exclusión (LockService)...");

      const payload = {
        id_booking: document.getElementById('ia-booking').value,
        remitente: document.getElementById('ia-remitente').value,
        direccion: document.getElementById('ia-direccion').value,
        cp: document.getElementById('ia-cp').value,
        piezas_estimadas: document.getElementById('ia-piezas').value,
        telefono: document.getElementById('ia-tel').value,
        chofer: document.getElementById('ia-chofer').value
      };

      google.script.run
        .withSuccessHandler((res) => {
          if (res.exito) {
            addLog(`ÉXITO: Booking despachado e inyectado con ID de rampa: ${res.id_pu}`);
            // Limpiar formulario de IA
            document.getElementById('ia-preview').classList.add('hidden');
            document.getElementById('btn-despacho').classList.add('hidden');
            document.getElementById('file-picker').value = '';
            actualizarTodo();
          } else {
            addLog("Error de inyección dual: " + res.error, true);
          }
        })
        .withFailureHandler((err) => {
          addLog("Fallo crítico de despacho: " + err.message, true);
        })
        .ejecutarDespachoBookingACampo(JSON.stringify(payload));
    }
  </script>
